function rayBox(origin,dir,lo,hi){let near=0,far=50;for(let i=0;i<3;i++){if(Math.abs(dir[i])<1e-7){if(origin[i]<lo[i]||origin[i]>hi[i])return Infinity}else{let a=(lo[i]-origin[i])/dir[i],b=(hi[i]-origin[i])/dir[i];near=Math.max(near,Math.min(a,b));far=Math.min(far,Math.max(a,b))}}return near<=far?near:Infinity}
function rayTriangle(o,d,a,b,c){let sub=(x,y)=>x.map((v,i)=>v-y[i]),cross=(x,y)=>[x[1]*y[2]-x[2]*y[1],x[2]*y[0]-x[0]*y[2],x[0]*y[1]-x[1]*y[0]],dot=(x,y)=>x.reduce((n,v,i)=>n+v*y[i],0),e1=sub(b,a),e2=sub(c,a),p=cross(d,e2),det=dot(e1,p);if(Math.abs(det)<1e-8)return Infinity;let t=sub(o,a),u=dot(t,p)/det;if(u<0||u>1)return Infinity;let q=cross(t,e1),v=dot(d,q)/det;if(v<0||u+v>1)return Infinity;let hit=dot(e2,q)/det;return hit>0?hit:Infinity}
